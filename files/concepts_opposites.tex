\section*{Conceptual Notes}

\subsection*{Relativity and Spacetime}
\begin{itemize}
\item \textbf{Galilean Relativity}: Framework where time is absolute and space is Euclidean
\item \textbf{Newtonian Absolute Space}: Non-relativistic framework with fixed spatial coordinates
\item \textbf{Three-Velocity}: Classical velocity vector without time component
\end{itemize}

\subsection*{Thermodynamics and Information}
\begin{itemize}
\item \textbf{Negentropy}: Measure of distance from thermodynamic equilibrium
\item \textbf{Non-equilibrium Thermodynamics}: Systems with energy/matter flows
\item \textbf{Deterministic Information}: Perfectly predictable signals with zero uncertainty
\end{itemize}

\subsection*{Probability and Decision Theory}
\begin{itemize}
\item \textbf{Dirac Delta Function}: Represents definite, certain outcomes
\item \textbf{Frequentist Inference}: Probability as long-run frequency rather than belief
\item \textbf{Disutility Function}: Measures costs rather than benefits
\end{itemize}

\subsection*{Complex Systems}
\begin{itemize}
\item \textbf{Reductionism}: Understanding systems through constituent parts
\item \textbf{Linear Dynamics}: Systems with proportional cause-effect relationships
\item \textbf{Static Equilibrium}: Systems without evolutionary change
\end{itemize}
